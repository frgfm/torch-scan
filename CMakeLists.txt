cmake_minimum_required(VERSION 3.19)
project(torchscan LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(Threads REQUIRED)

# Query the installed wheel without loading cpp_extension/setuptools or requiring a CUDA toolkit.
execute_process(COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/_build_metadata.py"
  OUTPUT_VARIABLE TORCH_CONFIG OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(JSON TORCH_ROOT GET "${TORCH_CONFIG}" root)
file(READ "${TORCH_ROOT}/share/cmake/Torch/TorchConfig.cmake" TORCH_CMAKE)
string(REGEX MATCH "(CXX_STANDARD[ \t\n]+|cxx_std_)([0-9]+)" TORCH_STANDARD "${TORCH_CMAKE}")
if(NOT TORCH_STANDARD)
  message(FATAL_ERROR "Cannot determine the installed PyTorch C++ standard")
endif()
set(TORCH_STANDARD "${CMAKE_MATCH_2}")

Python_add_library(_torchscan_native MODULE WITH_SOABI native/collector.cpp)
target_include_directories(_torchscan_native PRIVATE "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include")
set_target_properties(_torchscan_native PROPERTIES CXX_STANDARD "${TORCH_STANDARD}" CXX_STANDARD_REQUIRED ON)
target_compile_definitions(_torchscan_native PRIVATE TORCH_API_INCLUDE_EXTENSION_H TORCH_EXTENSION_NAME=_torchscan_native)
target_link_libraries(_torchscan_native PRIVATE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  string(JSON TORCH_ABI GET "${TORCH_CONFIG}" abi)
  target_compile_definitions(_torchscan_native PRIVATE "_GLIBCXX_USE_CXX11_ABI=${TORCH_ABI}")
endif()
if(NOT WIN32)
  foreach(NAME COMPILER_TYPE STDLIB BUILD_ABI)
    string(JSON KIND TYPE "${TORCH_CONFIG}" pybind ${NAME})
    if(NOT KIND STREQUAL "NULL")
      string(JSON VALUE GET "${TORCH_CONFIG}" pybind ${NAME})
      target_compile_definitions(_torchscan_native PRIVATE "PYBIND11_${NAME}=\"${VALUE}\"")
    endif()
  endforeach()
endif()
foreach(NAME c10 torch torch_cpu torch_python)
  find_library(TORCH_${NAME} NAMES ${NAME} PATHS "${TORCH_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(_torchscan_native PRIVATE "${TORCH_${NAME}}")
endforeach()
if(APPLE)
  set_target_properties(_torchscan_native PROPERTIES INSTALL_RPATH "@loader_path/torch/lib")
elseif(UNIX)
  set_target_properties(_torchscan_native PROPERTIES INSTALL_RPATH "$ORIGIN/torch/lib")
endif()
if(MSVC)
  target_compile_options(_torchscan_native PRIVATE /utf-8 /EHsc)
endif()
install(TARGETS _torchscan_native LIBRARY DESTINATION . RUNTIME DESTINATION .)

add_subdirectory(native/count)
Python_add_library(_analysis MODULE WITH_SOABI native/count/python.cpp)
target_link_libraries(_analysis PRIVATE torchscan_count_core)
install(TARGETS _analysis LIBRARY DESTINATION torchscan RUNTIME DESTINATION torchscan)
