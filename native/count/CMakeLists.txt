cmake_minimum_required(VERSION 3.18)
project(torchscan_count LANGUAGES CXX)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
find_path(BOOST_INCLUDE_DIR boost/multiprecision/cpp_int.hpp REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/rules.json" RULES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ignored.json" IGNORED)
configure_file(tables.hpp.in tables.hpp @ONLY)
add_executable(torchscan-count count.cpp formulas.cpp main.cpp)
target_compile_features(torchscan-count PRIVATE cxx_std_17)
target_include_directories(torchscan-count PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${NLOHMANN_JSON_INCLUDE_DIR}" "${BOOST_INCLUDE_DIR}")
if(MSVC)
  target_compile_options(torchscan-count PRIVATE /utf-8)
endif()
install(TARGETS torchscan-count RUNTIME DESTINATION bin)
install(FILES ../../LICENSE THIRD_PARTY_LICENSES.txt DESTINATION share/licenses/torchscan-count)
