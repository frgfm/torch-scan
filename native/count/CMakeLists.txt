cmake_minimum_required(VERSION 3.19)
project(torchscan_count LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/dependencies.cmake")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/rules.json" RULES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ignored.json" IGNORED)
configure_file(tables.hpp.in tables.hpp @ONLY)
add_library(torchscan_count_core OBJECT count.cpp formulas.cpp)
set_target_properties(torchscan_count_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(torchscan_count_core PUBLIC cxx_std_17)
target_include_directories(torchscan_count_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" "${INCLUDE_DIR}")
add_executable(torchscan-count main.cpp)
target_link_libraries(torchscan-count PRIVATE torchscan_count_core)
if(MSVC)
  target_compile_options(torchscan_count_core PUBLIC /utf-8 /EHsc)
endif()
if(SKBUILD)
  install(TARGETS torchscan-count RUNTIME DESTINATION "${SKBUILD_SCRIPTS_DIR}")
else()
  install(TARGETS torchscan-count RUNTIME DESTINATION bin)
  install(FILES ../../LICENSE THIRD_PARTY_LICENSES.txt DESTINATION share/licenses/torchscan-count)
endif()
